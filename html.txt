    <script>
        // 1. Initial Loading Logic
        let progress = 0;
        const progressFill = document.getElementById('progress-fill');
        const progressText = document.getElementById('progress-text');

        const interval = setInterval(() => {
            progress += 1;
            progressFill.style.width = progress + '%';
            progressText.innerText = progress + '%';

            if (progress >= 100) {
                clearInterval(interval);
                document.getElementById('loader-screen').style.display = 'none';
                document.getElementById('login-screen').style.display = 'block';
            }
        }, 30);

        // 2. Login Logic
        function handleLogin() {
            const user = document.getElementById('username').value;
            const pass = document.getElementById('password').value;
            const msg = document.getElementById('login-msg');

            if (user === 'y2j2026' && pass === '10111') {
                msg.innerText = "Authenticating... please wait 3-5 seconds";
                msg.style.color = "var(--primary)";
                
                setTimeout(() => {
                    showScreen('dashboard');
                    startClock();
                }, 4000);
            } else {
                msg.innerText = "Invalid credentials!";
                msg.style.color = "red";
            }
        }

        // 3. Navigation
        function showScreen(screenId) {
            const screens = ['login-screen', 'dashboard', 'balance-screen', 'withdrawal-screen', 'arrears-screen'];
            screens.forEach(s => document.getElementById(s).style.display = 'none');
            document.getElementById(screenId).style.display = 'block';
            
            // Reset withdrawal state if going back
            if(screenId !== 'withdrawal-screen') {
                document.getElementById('withdraw-loading').style.display = 'none';
                document.getElementById('decline-msg').style.display = 'none';
            }
        }

        // 4. Live Clock
        function startClock() {
            setInterval(() => {
                const now = new Date();
                document.getElementById('clock-display').innerText = now.toLocaleString();
            }, 1000);
        }

        // 5. Withdrawal Logic (The 98% Decline)
        function processWithdrawal() {
            const holder = document.getElementById('w-holder').value;
            const bank = document.getElementById('w-bank').value;
            
            if(!holder || !bank) {
                alert("Please fill in banking details");
                return;
            }

            const loadDiv = document.getElementById('withdraw-loading');
            const fill = document.getElementById('withdraw-fill');
            const status = document.getElementById('withdraw-status');
            const decline = document.getElementById('decline-msg');

            loadDiv.style.display = 'block';
            decline.style.display = 'none';
            let wProgress = 0;

            const wInterval = setInterval(() => {
                wProgress += 2;
                fill.style.width = wProgress + '%';
                status.innerText = wProgress + '%';

                if (wProgress >= 98) {
                    clearInterval(wInterval);
                    decline.style.display = 'block';
                    status.innerText = "98% - System Error";
                }
            }, 50);
        }
    </script>
